clc;
close all;
clear all;

T = 2*pi;
omega = 2*pi/T;

t = linspace(0,4*pi,1000);

N = 1000; % Harmonics

% Fourier Series Rectangular Wave
sq_wave = zeros(size(t));

figure();

for n = 1:2:N
    sq_wave = sq_wave + 4/(n*pi)*sin(n*omega*t);

    plot(t,sq_wave);
    xlabel("Time");
    ylabel("Amplitude");
    title("Fourier Series Rectangular Wave");
    grid on;
    hold on;

    pause(2);
end

figure();
plot(t,sq_wave);
xlabel("Time");
ylabel("Amplitude");
title("Fourier Series Rectangular Wave");
grid on;
