
clc;
clear all;
close all;

% Z-Transform
syms n;

x = sin(n);

f = ztrans(x);

disp('Z-transform of sin(n):');
pretty(f);

% Inverse Z-Transform
inversex = iztrans(f);

disp('Inverse Z-transform:');
pretty(inversex);

% Pole-Zero Plot
figure;

pz = tf([1],[1,2,1]);

pzmap(pz);

grid on;
