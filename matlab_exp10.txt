
clc;
clear;
close all;

%% 1. Z-Transform

syms n z

X1 = ztrans((1/2)^n,n,z);

X2 = ztrans(2^n,n,z);

X3 = ztrans(n*(1/2)^n,n,z);

disp('Z transforms:');

disp([X1; X2; X3]);

disp('Inverse Z-transform:');

disp(iztrans(X1,z,n));


%% 2. Pole-Zero Plot

b = [1 -0.25];

a = conv([1 -0.5],[1 -0.8]);

disp('Zeros:');
disp(roots(b).');

disp('Poles:');
disp(roots(a).');

figure;
zplane(b,a);
grid on;
title('Pole-Zero Plot');


%% 3. ROC

g = linspace(-3,3,400);

[X,Y] = meshgrid(g,g);

R = hypot(X,Y);

th = linspace(0,2*pi,300);

m = {R>2, R<0.5, R>0.5 & R<2};

titles = {'|z| > 2','|z| < 0.5','0.5 < |z| < 2'};

figure;

for k = 1:3

    subplot(1,3,k);
    hold on;

    imagesc(g,g,double(m{k}));

    colormap([1 1 1; 0.65 0.82 1]);

    plot(cos(th),sin(th),'k--');

    plot(0.5*cos(th),0.5*sin(th),'r--');
    plot(2*cos(th),2*sin(th),'r--');

    plot(0.5,0,'rx','MarkerSize',10,'LineWidth',2);
    plot(2,0,'rx','MarkerSize',10,'LineWidth',2);

    plot(0,0,'bo');

    set(gca,'YDir','normal');

    axis equal;
    axis([-3 3 -3 3]);
    grid on;

    title(titles{k});
    xlabel('Real(z)');
    ylabel('Imag(z)');

end

sgtitle('Regions of Convergence');


%% 4. Frequency Response

w = linspace(0,pi,512);

H = (1-0.25*exp(-1j*w))./ ...
    ((1-0.5*exp(-1j*w)).*(1-0.8*exp(-1j*w)));

figure;

subplot(2,1,1);
plot(w/pi,abs(H));
grid on;
title('Magnitude Response');
xlabel('\omega/\pi');
ylabel('|H|');

subplot(2,1,2);
plot(w/pi,unwrap(angle(H)));
grid on;
title('Phase Response');
xlabel('\omega/\pi');
ylabel('Phase (rad)');


%% 5. Stability and Impulse Response

p = roots(a);

if all(abs(p)<1)

    disp('Causal system: STABLE (all poles inside unit circle)');

else

    disp('Causal system: UNSTABLE');

end

h = filter(b,a,[1 zeros(1,29)]);

figure;
stem(0:29,h,'filled');
grid on;
title('Impulse Response');
xlabel('n');
ylabel('h[n]');
